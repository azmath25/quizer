\quiztitle{Algebra}
\quizdescription{Equations, identities, roots, and numeric expressions.}
\quizsubject{Algebra}
\quizdifficulty{Foundation}
\quiztime{15}
\begin{problem}[points=2]
Solve $3x-7=14$.
\answer{7}
\begin{solution}Add $7$ and divide by $3$: $x=21/3=7$.\end{solution}
\end{problem}
\begin{problem}[points=2]
Which expression equals $(a+b)^2$?
\begin{choices}
\choice $a^2+b^2$
\correctchoice $a^2+2ab+b^2$
\choice $a^2+ab+b^2$
\choice $2a^2+2b^2$
\end{choices}
\begin{solution}Multiply $(a+b)(a+b)$ and collect like terms.\end{solution}
\end{problem}
\begin{problem}[points=3]
Find the positive solution of $x^2=18$. A radical or a decimal is fine.
\answer{3*sqrt(2)}
\begin{solution}$x=\sqrt{18}=3\sqrt{2}$.\end{solution}
\end{problem}
\begin{problem}[points=2]
If $f(x)=2x^2-1$, find $f(3)$.
\answer{17}
\begin{solution}$f(3)=2\cdot 9-1=17$.\end{solution}
\end{problem}
\begin{problem}[points=3]
What is the product of the two roots of $x^2-5x+6=0$?
\begin{choices}
\choice $-6$
\choice $-5$
\choice $5$
\correctchoice $6$
\end{choices}
\begin{solution}By Viète's formulas the product is the constant term, $6$. Alternatively, factor $(x-2)(x-3)$.\end{solution}
\end{problem}
\begin{problem}[points=3]
Simplify $\frac{2}{3}+\frac{5}{6}$. Enter a fraction or a decimal.
\answer{3/2}
\begin{solution}$\frac{2}{3}+\frac{5}{6}=\frac{4+5}{6}=\frac{3}{2}$.\end{solution}
\end{problem}
